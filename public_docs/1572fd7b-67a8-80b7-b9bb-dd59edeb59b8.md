#### docker install

<a id="b-1572fd7b67a8803ab2fdd85815c6398e"></a>

<pre><code class="language-sh">curl https://get.docker.com | sh&#10;sudo usermod -aG docker $USER&#10;sudo reboot&#10;sudo systemctl start docker&#10;sudo systemctl enable docker&#10;docker --version</code></pre>

<a id="b-1572fd7b67a88073939cff332ef8f289"></a>

#### docker compose

<a id="b-1572fd7b67a88053bcaff580090ec857"></a>

<pre><code class="language-sh">sudo curl -L https://github.com/docker/compose/releases/download/1.16.1/docker-compose-`uname -s`-`uname -m` -o /usr/local/bin/docker-compose&#10;sudo chmod +x /usr/local/bin/docker-compose&#10;docker-compose --version</code></pre>

<a id="b-1572fd7b67a88051849feca461d84f97"></a>

#### minecraft

<a id="b-1572fd7b67a880389768e6216a9e87ae"></a>

<pre><code class="language-sh">mkdir -p minecraft/data&#10;vim minecraft/docker-compose.yml</code></pre>
